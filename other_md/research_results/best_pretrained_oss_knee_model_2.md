class OfflineSAMBackbone(nn.Module):
    def __init__(self, checkpoint_path: str, img_size: int = 512):
        super().__init__()
        sam = sam_model_registry["vit_b"](checkpoint=checkpoint_path)
        self.encoder = sam.image_encoder

        # Interpolate positional embeddings if resolution differs from 1024x1024
        if img_size != 1024:
            target_patches = img_size // 16
            orig_pos = self.encoder.pos_embed.data.permute(0, 3, 1, 2)  # [1, 768, 64, 64]
            resampled_pos = nn.functional.interpolate(
                orig_pos, size=(target_patches, target_patches), 
                mode="bicubic", align_corners=False
            )
            self.encoder.pos_embed = nn.Parameter(resampled_pos.permute(0, 2, 3, 1))
            self.encoder.img_size = img_size

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        # Input shape: [B, 3, img_size, img_size]
        features = self.encoder(x)  # [B, 256, img_size//16, img_size//16]
        return features.mean(dim=[-2, -1])  # Returns [B, 256] feature vectors
